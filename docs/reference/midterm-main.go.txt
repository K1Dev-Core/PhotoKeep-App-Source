package main

import (
	"bufio"
	"fmt"
	"io"
	"os"
	"path/filepath"
	"runtime"
	"strings"
	"sync"
	"time"

	"github.com/glebarez/sqlite"
	"gorm.io/gorm"
	"gorm.io/gorm/logger"
)

const databaseFile = "crossplatform.db"

type FileRecord struct {
	Idx      uint   `gorm:"column:idx;primaryKey;autoIncrement"`
	Dest     string `gorm:"column:dest;not null"`
	Filename string `gorm:"column:filename;not null"`
	Status   string `gorm:"column:status;not null;default:backed_up"`
}

func (FileRecord) TableName() string { return "files" }

type Setting struct {
	Idx    uint   `gorm:"column:idx;primaryKey;autoIncrement"`
	Source string `gorm:"column:source;not null"`
	Dest   string `gorm:"column:dest;not null"`
}

func (Setting) TableName() string { return "settings" }

type App struct {
	db     *gorm.DB
	source string
	dest   string
}

func main() {
	db, err := gorm.Open(sqlite.Open(databaseFile), &gorm.Config{Logger: logger.Default.LogMode(logger.Silent)})
	if err != nil {
		fmt.Println("เปิดฐานข้อมูลไม่สำเร็จ:", err)
		return
	}
	if err = db.AutoMigrate(&FileRecord{}, &Setting{}); err != nil {
		fmt.Println("เตรียมฐานข้อมูลไม่สำเร็จ:", err)
		return
	}
	a := &App{db: db}
	if err := a.loadSettings(); err != nil {
		fmt.Println("อ่าน setting ไม่สำเร็จ:", err)
		return
	}
	printBanner()
	if a.dest == "" {
		fmt.Println("dest เป็นค่าว่าง กรุณาตั้งค่าด้วย /dest <path>")
	}
	scanner := bufio.NewScanner(os.Stdin)
	for {
		fmt.Print("> ")
		if !scanner.Scan() || a.handle(strings.TrimSpace(scanner.Text())) {
			break
		}
	}
	fmt.Println("ออกจากโปรแกรม")
}

func (a *App) loadSettings() error {
	var setting Setting
	if err := a.db.Order("idx desc").First(&setting).Error; err != nil {
		if err == gorm.ErrRecordNotFound {
			return nil
		}
		return fmt.Errorf("normalize settings: %w", err)
	}
	a.source, a.dest = setting.Source, setting.Dest
	return nil
}

func (a *App) handle(line string) bool {
	if line == "" {
		return false
	}
	parts := strings.Fields(line)
	cmd := strings.ToLower(parts[0])
	raw := strings.TrimSpace(strings.TrimPrefix(line, parts[0]))
	switch cmd {
	case "/exit":
		return true
	case "/help":
		a.help()
	case "/source":
		a.setSource(raw)
	case "/dest":
		a.setDest(raw)
	case "/set":
		a.setBoth(raw)
	case "/settings":
		a.settings()
	case "/list":
		a.list(raw)
	case "/add":
		a.add(raw)
	case "/move":
		a.move(raw)
	case "/check":
		a.check()
	case "/delete":
		a.deleteFiles(raw)
	case "/clean":
		a.clean()
	default:
		fmt.Printf("ไม่รู้จักคำสั่ง: %s\n", cmd)
	}
	return false
}

func cleanPath(raw string) string { return strings.Trim(strings.TrimSpace(raw), "\"'") }

func (a *App) saveSetting(source, dest string) error {
	row := Setting{Source: source, Dest: dest}
	if err := a.db.Create(&row).Error; err != nil {
		return err
	}
	a.source, a.dest = source, dest
	return nil
}

func (a *App) setSource(raw string) {
	path := cleanPath(raw)
	if path == "" {
		fmt.Println("วิธีใช้: //source <path>")
		return
	}
	if err := a.saveSetting(path, a.dest); err != nil {
		fmt.Println("บันทึก source ไม่สำเร็จ:", err)
		return
	}
	var row Setting
	a.db.Order("idx desc").First(&row)
	fmt.Printf("บันทึก source แล้ว idx=%d source=%s\n", row.Idx, path)
}

func (a *App) setDest(raw string) {
	path := cleanPath(raw)
	if path == "" {
		fmt.Println("วิธีใช้: /dest <path>")
		return
	}
	if err := a.saveSetting(a.source, path); err != nil {
		fmt.Println("บันทึก dest ไม่สำเร็จ:", err)
		return
	}
	var row Setting
	a.db.Order("idx desc").First(&row)
	fmt.Printf("บันทึก dest แล้ว idx=%d dest=%s\n", row.Idx, path)
	a.check()
}

func (a *App) setBoth(raw string) {
	parts := strings.Fields(raw)
	if len(parts) < 2 {
		fmt.Println("วิธีใช้: /set <source> <dest>")
		return
	}
	source, dest := cleanPath(parts[0]), cleanPath(strings.Join(parts[1:], " "))
	if err := a.saveSetting(source, dest); err != nil {
		fmt.Println("บันทึก setting ไม่สำเร็จ:", err)
		return
	}
	var row Setting
	a.db.Order("idx desc").First(&row)
	fmt.Printf("บันทึกตั้งค่าแล้ว idx=%d source=%s dest=%s\n", row.Idx, source, dest)
}

func (a *App) settings() {
	var rows []Setting
	if err := a.db.Order("idx asc").Find(&rows).Error; err != nil {
		fmt.Println("อ่านตั้งค่าไม่สำเร็จ:", err)
		return
	}
	if len(rows) == 0 {
		fmt.Println("ยังไม่มีข้อมูลตั้งค่า")
		return
	}
	for _, row := range rows {
		fmt.Printf("idx=%d source=%s dest=%s\n", row.Idx, row.Source, row.Dest)
	}
}

func (a *App) list(raw string) {
	parts := strings.Fields(raw)
	if len(parts) == 0 {
		fmt.Println("วิธีใช้: /list source | dest | db [path]")
		return
	}
	switch strings.ToLower(parts[0]) {
	case "source":
		a.listFolder(a.source, "source")
	case "dest":
		a.listFolder(a.dest, "dest")
	case "db":
		path := a.dest
		if len(parts) > 1 {
			path = cleanPath(strings.TrimSpace(strings.TrimPrefix(raw, parts[0])))
		}
		if path == "" {
			fmt.Println("dest เป็นค่าว่าง กรุณาตั้งค่าด้วย /dest <path>")
			return
		}
		var rows []FileRecord
		if err := a.db.Where("dest = ?", path).Order("idx asc").Find(&rows).Error; err != nil {
			fmt.Println("อ่านตาราง files ไม่สำเร็จ:", err)
			return
		}
		if len(rows) == 0 {
			fmt.Println("ยังไม่มีประวัติของ dest:", path)
			return
		}
		for _, row := range rows {
			fmt.Printf("idx=%d dest=%s filename=%s\n", row.Idx, row.Dest, row.Filename)
		}
	default:
		fmt.Println("วิธีใช้: /list source | dest | db [path]")
	}
}

func (a *App) listFolder(dir, label string) {
	if dir == "" {
		fmt.Printf("%s เป็นค่าว่าง กรุณาตั้งค่าด้วย /%s <path>\n", label, label)
		return
	}
	entries, err := os.ReadDir(dir)
	if err != nil {
		fmt.Printf("อ่านไฟล์ใน %s ไม่สำเร็จ: %v\n", label, err)
		return
	}
	count := 0
	for _, entry := range entries {
		if entry.Type().IsRegular() {
			fmt.Println(entry.Name())
			count++
		}
	}
	if count == 0 {
		fmt.Printf("ไม่มีไฟล์ใน %s: %s\n", label, dir)
	}
}

func names(raw string) []string {
	raw = strings.TrimSpace(raw)
	if raw == "" {
		return nil
	}
	parts := strings.Split(raw, ",")
	if len(parts) == 1 {
		parts = strings.Fields(raw)
	}
	result := make([]string, 0, len(parts))
	for _, part := range parts {
		if part = cleanPath(part); part != "" {
			result = append(result, filepath.Base(part))
		}
	}
	return result
}

func (a *App) add(raw string) {
	parts := strings.Fields(raw)
	if len(parts) < 2 {
		fmt.Println("วิธีใช้: /add <dest> <filename>")
		return
	}
	dest, filename := cleanPath(parts[0]), filepath.Base(cleanPath(strings.Join(parts[1:], " ")))
	row := FileRecord{Dest: dest, Filename: filename, Status: "backed_up"}
	if err := a.db.Create(&row).Error; err != nil {
		fmt.Println("บันทึกไม่สำเร็จ:", err)
		return
	}
	fmt.Printf("บันทึกแล้ว idx=%d dest=%s filename=%s\n", row.Idx, dest, filename)
}

func (a *App) move(raw string) {
	if a.source == "" {
		fmt.Println("source เป็นค่าว่าง กรุณาตั้งค่าด้วย /source <path>")
		return
	}
	if a.dest == "" {
		fmt.Println("dest เป็นค่าว่าง กรุณาตั้งค่าด้วย /dest <path>")
		return
	}
	selected := names(raw)
	if strings.EqualFold(strings.TrimSpace(raw), "all") {
		selected = nil
		entries, err := os.ReadDir(a.source)
		if err != nil {
			fmt.Println("อ่านไฟล์ใน source ไม่สำเร็จ:", err)
			return
		}
		for _, entry := range entries {
			if entry.Type().IsRegular() {
				selected = append(selected, entry.Name())
			}
		}
	}
	if len(selected) == 0 {
		fmt.Println("วิธีใช้: /move <file>,... หรือ all")
		return
	}
	fmt.Println()
	fmt.Println("┌─ กำลังย้ายไฟล์ ───────────────────────────────────────────────────────────────┐")
	fmt.Printf("│ ต้นทาง: %-68s│\n", a.source)
	fmt.Printf("│ ปลายทาง: %-67s│\n", a.dest)
	fmt.Println("└───────────────────────────────────────────────────────────────────────────────┘")
	startedAt := time.Now()
	moved := 0
	var existingRows []FileRecord
	if err := a.db.Where("dest = ? AND (status = ? OR status = '')", a.dest, "backed_up").Find(&existingRows).Error; err != nil {
		fmt.Println("อ่านข้อมูลซ้ำจากฐานข้อมูลไม่สำเร็จ:", err)
		return
	}
	existing := make(map[string]struct{}, len(existingRows))
	for _, row := range existingRows {
		existing[row.Filename] = struct{}{}
	}
	type moveResult struct {
		name   string
		moved  bool
		reason string
		err    error
	}
	jobs := make(chan string)
	results := make(chan moveResult)
	workerCount := runtime.NumCPU()
	if workerCount < 8 {
		workerCount = 8
	}
	if workerCount > 16 {
		workerCount = 16
	}
	if len(selected) < workerCount {
		workerCount = len(selected)
	}
	var workers sync.WaitGroup
	for i := 0; i < workerCount; i++ {
		workers.Add(1)
		go func() {
			defer workers.Done()
			for name := range jobs {
				src, dst := filepath.Join(a.source, name), filepath.Join(a.dest, name)
				if _, ok := existing[name]; ok {
					results <- moveResult{name: name, reason: "duplicate"}
					continue
				}
				if _, err := os.Stat(dst); err == nil {
					results <- moveResult{name: name, reason: "duplicate"}
					continue
				}
				if _, err := os.Stat(src); err != nil {
					results <- moveResult{name: name, reason: "missing"}
					continue
				}
				if err := moveFile(src, dst); err != nil {
					results <- moveResult{name: name, err: err}
					continue
				}
				results <- moveResult{name: name, moved: true}
			}
		}()
	}
	go func() {
		for _, name := range selected {
			jobs <- name
		}
		close(jobs)
		workers.Wait()
		close(results)
	}()

	completed := 0
	for result := range results {
		completed++
		filled := completed * 20 / len(selected)
		fmt.Printf("\r[%s%s] %d/%d %d%% %s", strings.Repeat("#", filled), strings.Repeat("-", 20-filled), completed, len(selected), completed*100/len(selected), result.name)
		if result.reason == "duplicate" {
			fmt.Printf("\nมีไฟล์อยู่แล้วที่ dest ไม่ย้ายทับ: %s\n", result.name)
			continue
		}
		if result.reason == "missing" {
			fmt.Printf("\nไม่พบไฟล์ใน source: %s\n", result.name)
			continue
		}
		if result.err != nil {
			fmt.Printf("\nย้าย %s ไม่สำเร็จ: %v\n", result.name, result.err)
			continue
		}
		if result.moved {
			moved++
			if err := a.db.Create(&FileRecord{Dest: a.dest, Filename: result.name, Status: "backed_up"}).Error; err != nil {
				fmt.Printf("\nบันทึก %s ไม่สำเร็จ: %v\n", result.name, err)
			}
		}
	}
	fmt.Println()
	finishedAt := time.Now()
	fmt.Println("┌─ สรุปผลการย้ายไฟล์ ────────────────────────────────────────────────────────────┐")
	fmt.Printf("│ ย้ายไฟล์สำเร็จ: %d ไฟล์\n", moved)
	fmt.Printf("│ เริ่ม: %s\n", startedAt.Format("02/01/2006 15:04:05.000"))
	fmt.Printf("│ สิ้นสุด: %s\n", finishedAt.Format("02/01/2006 15:04:05.000"))
	durationMS := float64(finishedAt.Sub(startedAt).Microseconds()) / 1000
	if durationMS >= 1000 {
		fmt.Printf("│ ใช้เวลา: %.4f วินาที\n", durationMS/1000)
	} else {
		fmt.Printf("│ ใช้เวลา: %.4f ms\n", durationMS)
	}
	fmt.Println("└───────────────────────────────────────────────────────────────────────────────┘")
}

func (a *App) check() {
	if a.dest == "" {
		fmt.Println("dest เป็นค่าว่าง กรุณาตั้งค่าด้วย /dest <path>")
		return
	}
	var rows []FileRecord
	if err := a.db.Where("dest = ? AND (status = ? OR status = '')", a.dest, "backed_up").Find(&rows).Error; err != nil {
		fmt.Println("ตรวจสอบ dest ของ", a.dest, "ไม่สำเร็จ:", err)
		return
	}
	entries, err := os.ReadDir(a.dest)
	if err != nil {
		fmt.Println("อ่านไฟล์ใน dest ไม่สำเร็จ:", err)
		return
	}
	actual, dbFiles := map[string]bool{}, map[string]bool{}
	for _, entry := range entries {
		if entry.Type().IsRegular() {
			actual[entry.Name()] = true
		}
	}
	for _, row := range rows {
		dbFiles[row.Filename] = true
		if !actual[row.Filename] {
			fmt.Printf("มีในฐานข้อมูลแต่ไม่มีใน dest: %s\n", row.Filename)
		}
	}
	for filename := range actual {
		if !dbFiles[filename] {
			fmt.Printf("มีใน dest แต่ไม่มีในฐานข้อมูล: %s\n", filename)
		}
	}
	if len(rows) == len(actual) {
		fmt.Printf("ไฟล์ใน dest ตรงกับฐานข้อมูล (%d ไฟล์)\n", len(rows))
	}
}

func (a *App) deleteFiles(raw string) {
	if a.dest == "" {
		fmt.Println("dest เป็นค่าว่าง กรุณาตั้งค่าด้วย /dest <path>")
		return
	}
	if strings.HasPrefix(strings.ToLower(strings.TrimSpace(raw)), "dest ") {
		raw = strings.TrimSpace(strings.TrimSpace(raw)[5:])
	}
	selected := names(raw)
	if strings.EqualFold(strings.TrimSpace(raw), "all") {
		entries, _ := os.ReadDir(a.dest)
		for _, entry := range entries {
			if entry.Type().IsRegular() {
				selected = append(selected, entry.Name())
			}
		}
	}
	if len(selected) == 0 {
		fmt.Println("วิธีใช้: /delete <file>,... หรือ all")
		return
	}
	deleted := 0
	for _, name := range selected {
		if err := os.Remove(filepath.Join(a.dest, name)); err != nil {
			if os.IsNotExist(err) {
				fmt.Printf("ไม่พบไฟล์ใน dest: %s\n", name)
			} else {
				fmt.Printf("ลบ %s ไม่สำเร็จ: %v\n", name, err)
			}
			continue
		}
		if err := a.db.Model(&FileRecord{}).
			Where("dest = ? AND filename = ? AND (status = ? OR status = '')", a.dest, name, "backed_up").
			Updates(map[string]interface{}{"status": "deleted"}).Error; err != nil {
			fmt.Println("ลบข้อมูลไม่สำเร็จ:", err)
		}
		deleted++
	}
	fmt.Printf("ลบไฟล์แล้ว %d ไฟล์ และลบจากฐานข้อมูลแล้ว\n", deleted)
}

func (a *App) clean() {
	if err := a.db.Session(&gorm.Session{AllowGlobalUpdate: true}).Delete(&FileRecord{}).Error; err != nil {
		fmt.Println("ลบข้อมูลในฐานข้อมูลไม่สำเร็จ:", err)
		return
	}
	if err := a.db.Session(&gorm.Session{AllowGlobalUpdate: true}).Delete(&Setting{}).Error; err != nil {
		fmt.Println("ลบข้อมูลไม่สำเร็จ:", err)
		return
	}
	a.source, a.dest = "", ""
	fmt.Println("ลบข้อมูลในตาราง files และ settings แล้ว")
}

func moveFile(src, dst string) error {
	if err := os.Rename(src, dst); err == nil {
		return nil
	}
	in, err := os.Open(src)
	if err != nil {
		return err
	}
	out, err := os.Create(dst)
	if err != nil {
		_ = in.Close()
		return err
	}
	if _, err = io.Copy(out, in); err != nil {
		out.Close()
		_ = in.Close()
		_ = os.Remove(dst)
		return err
	}
	if err = out.Close(); err != nil {
		_ = in.Close()
		_ = os.Remove(dst)
		return err
	}
	if err = in.Close(); err != nil {
		_ = os.Remove(dst)
		return err
	}
	return os.Remove(src)
}
