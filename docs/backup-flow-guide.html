เลือกโฟลเดอร์ กดสำรอง
แล้ว PhotoKeep ทำอะไรบ้าง

คู่มือนี้ไล่ตั้งแต่กำหนดโฟลเดอร์ต้นทางและปลายทาง จนแอปตรวจรูป คัดลอกลงฐานข้อมูล และรายงานผลเมื่อเสร็จ โดยอ้างอิงไฟล์ที่แอปใช้งานจริง

เลือก pathตรวจสอบและบันทึกสแกนรูปสำรองพร้อมกันสูงสุด 4 งานแสดงความคืบหน้า

แผนภาพการสำรอง

กดเล่นเพื่อดูทีละขั้น หรือเลือกหมายเลขด้านล่างเพื่อข้ามไปยังขั้นนั้น

ภาพจำลองเพื่ออธิบาย
ขั้นที่ 1 จาก 4เลือก path ของรูปต้นฉบับและโฟลเดอร์ที่จะสำรอง
1

เลือกโฟลเดอร์

เลือกที่เก็บรูปต้นฉบับและตำแหน่งสำรอง พร้อมเลือกว่าจะเก็บต้นฉบับไว้หรือย้ายออกหลังสำรอง

2

ตรวจและบันทึกค่า

แอปตรวจว่า path ใช้ได้ โฟลเดอร์ไม่ซ้อนกัน และสร้างโฟลเดอร์ปลายทางก่อนบันทึกการตั้งค่า

3

กด “สำรองรูป”

การบันทึก path ยังไม่เริ่มสำรอง ต้องกดปุ่มสำรอง จากนั้นหน้าต่างจะแสดงสถานะสแกนและคัดลอก

4

ทำงานจนเสร็จ

ตรวจรูปซ้ำ คัดลอกไฟล์ บันทึกข้อมูล และรายงานจำนวนเพิ่ม ข้าม และผิดพลาด พร้อมเวลาที่ใช้

ก่อนเริ่ม: ตั้งค่า path

เปิดหน้าตั้งค่าโฟลเดอร์ เลือกโฟลเดอร์ต้นทางและปลายทาง แล้วกำหนดว่าจะเก็บรูปต้นฉบับที่เดิมหรือไม่

โฟลเดอร์ต้นทาง

ตำแหน่งที่ PhotoKeep จะไล่ค้นหารูป รองรับ JPG, PNG, GIF, WebP, BMP, TIF และ TIFF โฟลเดอร์หรือไฟล์ที่ขึ้นต้นด้วยจุดจะถูกข้าม

โฟลเดอร์ปลายทาง

ตำแหน่งเก็บสำเนา PhotoKeep สร้างโฟลเดอร์นี้ให้ถ้ายังไม่มี และไม่อนุญาตให้ต้นทางกับปลายทางซ้อนกัน

“เก็บรูปต้นฉบับไว้ที่เดิม”

ค่าเริ่มต้นเปิดไว้: สำเนารูปไปยังปลายทางและคงต้นฉบับไว้ หากปิดไว้ ระบบจะลบต้นฉบับหลังสำรองและบันทึกข้อมูลสำเร็จ

กดบันทึก

ระบบตรวจ path และบันทึกการตั้งค่า การบันทึกนี้ยังไม่เริ่มสำรอง ต้องกลับไปกดปุ่ม “สำรองรูป” เพื่อเริ่มงาน

เมื่อกดสำรองแล้วเกิดอะไรขึ้น

งานแบ่งเป็นสแกนรายการรูปก่อน จากนั้น worker หลายตัวทำสำรอง โดยสถานะถูกส่งจาก Go ไปยังหน้าต่างแอปเป็นระยะ

① สแกนรายการ

เดินดูโฟลเดอร์ต้นทางแบบย่อยทุกชั้น ข้ามไฟล์ซ่อนและไฟล์ที่ไม่ใช่รูป แล้วนับจำนวนภาพที่จะทำงาน

② ตรวจสำเนาเดิม

เปรียบเทียบแหล่งที่มาและขนาดไฟล์ หากมีสำเนาเดิมจะตรวจ SHA-256 เพื่อข้ามไฟล์ที่เหมือนกัน ไม่เขียนซ้ำโดยไม่จำเป็น

③ คัดลอกและบันทึก

คัดลอกลงไฟล์ชั่วคราว ตรวจขนาดและ hash แล้วเปลี่ยนชื่อเป็นไฟล์ปลายทาง บันทึกรูปและข้อมูลไฟล์ลงฐานข้อมูล พร้อมสร้าง thumbnail

④ แสดงผล

แถบความคืบหน้าแสดงจำนวนที่เสร็จ กำลังทำ รอคิว และชื่อไฟล์ล่าสุด เมื่อครบจะแสดงจำนวนเพิ่ม/ข้าม/ผิดพลาดและเวลารวม

workers := minInt(4, maxInt(1, runtime.GOMAXPROCS(0))) จำนวนงานพร้อมกันจริง = ค่าที่น้อยกว่าระหว่าง 4 กับจำนวน CPU ที่ Go มองเห็น แต่จะมี worker อย่างน้อย 1 ตัว

func (a *Service) SaveBackupSettings(source, dest string, keepOriginal bool) (State, error) {
    source, dest = strings.TrimSpace(source), strings.TrimSpace(dest)
    if source == "" || dest == "" {
        return State{}, errors.New("กรุณาเลือกโฟลเดอร์ต้นทางและปลายทาง")
    }
    info, err := os.Stat(source)
    if err != nil || !info.IsDir() {
        return State{}, errors.New("โฟลเดอร์ต้นทางไม่มีอยู่หรือเปิดไม่ได้")
    }
    if err := os.MkdirAll(dest, 0755); err != nil {
        return State{}, fmt.Errorf("สร้างโฟลเดอร์ปลายทาง: %w", err)
    }
    if nested(source, dest) || nested(dest, source) {
        return State{}, errors.New("ต้นทางและปลายทางต้องเป็นคนละโฟลเดอร์และไม่ซ้อนกัน")
    }
    if err := a.db.Create(&Setting{Source: source, Dest: dest, KeepOriginal: &keepOriginal}).Error; err != nil {
        return State{}, err
    }
    return a.GetState()
}
